def observe(runner):

def normalize_image(value):
    result = {k: v for k, v in value.items() if k not in {"source", "destination", "path"}}
    if "mode" in result:
        result["mode"] = int(result["mode"], 8)
    if result["kind"] == "symlink":
        result["kind"] = "link"
    return result


def compare_baseline(plan, baseline):

class BoundExecution:
    def __init__(self, root, source, source_identity, source_check, *, runner=None, registry=DEFAULT_REGISTRY):

            self.readiness()

            compare_baseline(plan, baseline)
            require(observe(self.runner) == runtime and tool_pins() == policy["tools"], "preparation drift")
        return {**policy, "plan": plan}

    def existing_lock(self):
        path = external_paths(self.root, BEAD)["common"] / "gas-city-workflow/source-transition.lock"

                        dry_run=False, _session_transaction=txn,
                    ))
            finally:
                transaction.record["plan_sync_stdout"] = output.getvalue()
                transaction._save()

        def after():

            # An exact replay has already checked all accepted images. Keep it
            # read-only: validate the prior bound result, do not re-run Go or log.
            if transaction.record_path.exists():
                record = json.loads(transaction.record_path.read_text())
                if record.get("status") == "complete":
                    proof = record.get("postflight", {})
                    require(proof.get("command") == [GO, "test", "-json", "-count=1", "./test/docsync"]
                            and proof.get("returncode") == 0 and proof.get("tests_passed", 0) > 0,
                            "completed transaction lacks documentation proof")

            require(all(isinstance(event, dict) for event in events), "invalid documentation result event")

            before()
            check_external(self.root, plan)

def need(condition, message):

def git(source, *args):
    result = subprocess.run(["/usr/bin/git", "-C", str(source), *args], capture_output=True, timeout=30)
    need(result.returncode == 0, "source Git verification failed: " + args[0])
    return result.stdout


def source_binding(source, head, tree):
    # -I excludes cwd, user-site and PYTHON* startup overrides. Retain the
    # system site: the existing plan-sync helper needs distribution jsonschema.
    # Repository modules/caches remain separately verified and source-only.
    need(sys.flags.isolated == 1, "launch requires python3 -I")
    need(os.getuid() == os.geteuid() != 0, "run as the unprivileged owner")
    need(source in ALLOWED_SOURCES and source == source.resolve(), "source root is not approved/literal")
    need(re.fullmatch(r"[0-9a-f]{40}", head) and re.fullmatch(r"[0-9a-f]{40}", tree), "invalid source object identity")
    for key in os.environ:
        need(not key.startswith("GIT_CONFIG_") and key not in {
            "GIT_DIR", "GIT_COMMON_DIR", "GIT_WORK_TREE", "GIT_INDEX_FILE",
            "GIT_OBJECT_DIRECTORY", "GIT_ALTERNATE_OBJECT_DIRECTORIES", "GIT_REPLACE_REF_BASE",
        }, "unexpected Git environment override")
    need(git(source, "rev-parse", "--show-toplevel").decode().strip() == str(source), "source root is not exact Git root")
    need(git(source, "rev-parse", "HEAD").decode().strip() == head, "source HEAD drift")
    need(git(source, "rev-parse", "HEAD^{tree}").decode().strip() == tree, "source tree drift")
    need(git(source, "remote", "get-url", "origin").decode().strip() in {
        "https://github.com/loucmane/gas-city-operations.git", "git@github.com:loucmane/gas-city-operations.git",
    }, "wrong Operations source repository")
    git(source, "verify-commit", "--raw", head)
    git(source, "merge-base", "--is-ancestor", head, "refs/remotes/origin/main")
    need(git(source, "rev-parse", "refs/remotes/origin/main^{tree}").decode().strip() == tree,
         "source is not byte-identical to the locally bound merged tree")
    sys.dont_write_bytecode = True
    sys.pycache_prefix = os.devnull
    os.environ["PYTHONDONTWRITEBYTECODE"] = "1"
    os.environ["PYTHONPYCACHEPREFIX"] = os.devnull
    # Load the standard-library-only verifier FROM the verified Git object,
    # before importing any potentially changed repository module or cache.
    module = types.ModuleType("core_layout_integrity_bootstrap")
    module.__file__ = str(source / VERIFIER)
    exec(compile(git(source, "show", head + ":" + VERIFIER), module.__file__, "exec"), module.__dict__)
    module.reviewed_runtime(source, source)
    configs = {}
    for relative in CONFIGS:

def frozen_json(path, expected):
    need(re.fullmatch(r"[0-9a-f]{64}", expected), "invalid input SHA-256")
    need(path.is_absolute() and path == path.resolve(strict=True), "input path is not literal")
    info = path.lstat()
    need(stat.S_ISREG(info.st_mode) and info.st_nlink == 1 and info.st_uid == os.getuid()
         and info.st_size <= 16 * 1024 * 1024, "input type/owner/size refused")
    data = path.read_bytes()
    need(hashlib.sha256(data).hexdigest() == expected, "frozen input digest mismatch")
    def pairs(items):
        result = {}
        for key, value in items:
            need(key not in result, "duplicate JSON field")
            result[key] = value
        return result
    return json.loads(data, object_pairs_hook=pairs)
